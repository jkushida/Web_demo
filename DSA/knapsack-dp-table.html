擬似コード

dp[0][j] = 0 // 0行目は初期値
dp[i][0] = 0 // 容量0
for i = 1 ... N: // 品物を順に見る
skip = dp[i-1][j] // 入れない
if w[i] <= j: // 入れられるか
take = dp[i-1][j-w[i]] + v[i] // 入れる
dp[i][j] = max(skip, take) // 大きい方
else: dp[i][j] = skip // 入れない
